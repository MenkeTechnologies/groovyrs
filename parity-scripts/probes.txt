# `>>` on two closures is `Closure.andThen`: forward composition, receiver first.
def f = { it + 1 }; def g = { it * 2 }
println((f >> g)(3))
%%
# `<<` composes the other way round — the argument runs first.
def f = { it + 1 }; def g = { it * 2 }
println((f << g)(3))
%%
# Composition chains left-to-right, and each step is itself a closure.
def f = { it + 1 }; def g = { it * 2 }; def h = { it - 3 }
println((f >> g >> h)(3))
%%
# Composing two closure literals needs no name to hang the shape off.
println(({ it + 1 } >> { it * 2 })(3))
%%
# A composed closure keeps the arity of the one that runs first.
def add = { a, b -> a + b }; def show = { "=$it" }
println((add >> show)(2, 3))
%%
# `andThen`/`compose` are the method spellings of `>>`/`<<`.
def f = { it + 1 }; def g = { it * 2 }
println([f.andThen(g)(3), f.compose(g)(3), f.rightShift(g)(3), f.leftShift(g)(3)])
